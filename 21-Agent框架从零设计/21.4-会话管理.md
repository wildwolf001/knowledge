# 21.4 会话管理 —— SQLite/Redis/PostgreSQL 的持久化选型、Resume（恢复）机制

> 会话管理器是 Agent 框架的"档案管理员"——它负责记录 Agent 的每一次思考、每一步行动、每一个中间结果，并支持在任何时刻恢复之前的会话。没有会话管理，Agent 就是一个"做完即忘"的对话系统。

---

## 一、会话管理的核心职责

### 1.1 会话的生命周期

```
┌──────────┐     ┌──────────┐     ┌──────────┐     ┌──────────┐
│  CREATE   │────▶│  ACTIVE  │────▶│ PAUSED   │────▶│  ENDED   │
│  (新会话)  │     │ (运行中)  │     │ (暂停)    │     │ (正常/异常)│
└──────────┘     └──────────┘     └──────────┘     └──────────┘
                        │                                │
                        │  crash/restart                  │
                        ▼                                │
                  ┌──────────┐                            │
                  │ RESUME   │────────────────────────────┘
                  │ (恢复)    │   存档/清理
                  └──────────┘
```

一次完整的会话生命周期包含以下阶段：

1. **CREATE**：创建新会话，分配 Session ID，初始化上下文
2. **ACTIVE**：Agent 正在运行，每轮完成后持久化一次
3. **PAUSED**：会话因用户离开或系统重启而暂停
4. **RESUME**：从持久化存储中恢复之前的会话状态
5. **ENDED**：会话正常结束或异常终止，归档或清理

### 1.2 会话管理器需要存储什么？

```
Session
├── session_id        : 全局唯一标识
├── status            : active / paused / ended
├── created_at        : 创建时间
├── updated_at        : 最后活动时间
├── metadata          : 用户ID、上下文标签等
│
├── messages[]        : 完整的对话历史
│   ├── role          : system / user / assistant / tool
│   ├── content       : 消息内容
│   ├── tool_call     : 工具调用信息（可选）
│   └── timestamp     : 消息时间戳
│
├── turn_history[]    : 每轮 Turn 的详细记录
│   ├── turn_id
│   ├── thought
│   ├── action        : 工具调用
│   ├── observation   : 观察结果
│   └── latency_ms
│
├── state             : Agent 的运行时状态
│   ├── variables     : 中间变量
│   ├── current_step  : 当前执行到哪一步
│   └── memory_snapshot : 记忆摘要
└── config            : 创建会话时使用的配置
    ├── model
    ├── temperature
    └── system_prompt
```

## 二、三种存储引擎的实现

### 2.1 存储引擎通用接口

```python
# framework/session/base.py
"""会话存储引擎通用接口"""

from __future__ import annotations
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Any, Optional


class SessionStatus(str, Enum):
    ACTIVE = "active"
    PAUSED = "paused"
    ENDED = "ended"


@dataclass
class Session:
    session_id: str
    status: SessionStatus = SessionStatus.ACTIVE
    created_at: datetime = field(default_factory=datetime.now)
    updated_at: datetime = field(default_factory=datetime.now)
    messages: list[dict] = field(default_factory=list)
    turn_history: list[dict] = field(default_factory=list)
    state: dict = field(default_factory=dict)
    config: dict = field(default_factory=dict)
    metadata: dict = field(default_factory=dict)


class SessionStore(ABC):
    """会话存储引擎抽象基类"""
    
    @abstractmethod
    async def create(self, session_id: str, config: dict = None, 
                     metadata: dict = None) -> Session:
        """创建新会话"""
        ...
    
    @abstractmethod
    async def get(self, session_id: str) -> Optional[Session]:
        """获取会话"""
        ...
    
    @abstractmethod
    async def save(self, session: Session) -> None:
        """保存会话（全量保存）"""
        ...
    
    @abstractmethod
    async def append_message(self, session_id: str, message: dict) -> None:
        """追加一条消息到会话历史"""
        ...
    
    @abstractmethod
    async def update_status(self, session_id: str, status: SessionStatus) -> None:
        """更新会话状态"""
        ...
    
    @abstractmethod
    async def delete(self, session_id: str) -> None:
        """删除会话"""
        ...
    
    @abstractmethod
    async def list_sessions(self, limit: int = 100, offset: int = 0) -> list[Session]:
        """列出会话"""
        ...
```

### 2.2 SQLite 实现 —— 嵌入式持久化

SQLite 适合以下场景：
- 单机部署，不需要网络访问
- 开发环境、测试环境、小规模生产
- 数据量在 GB 级别以内
- 读写并发不高（SQLite 写操作是串行的）

```python
# framework/session/sqlite_store.py
"""SQLite 会话存储引擎"""

from __future__ import annotations
import json
import sqlite3
from datetime import datetime
from typing import Any, Optional
from uuid import uuid4

import aiosqlite

from .base import Session, SessionStatus, SessionStore


class SQLiteSessionStore(SessionStore):
    """SQLite 实现的会话存储
    
    优势：零配置，单文件，适合开发和单机部署
    局限：写并发低，不适合分布式场景
    """
    
    def __init__(self, db_path: str = "sessions.db"):
        self.db_path = db_path
    
    async def _init_db(self):
        """初始化数据库表结构"""
        async with aiosqlite.connect(self.db_path) as db:
            await db.execute("""
                CREATE TABLE IF NOT EXISTS sessions (
                    session_id TEXT PRIMARY KEY,
                    status TEXT NOT NULL DEFAULT 'active',
                    created_at TEXT NOT NULL,
                    updated_at TEXT NOT NULL,
                    messages TEXT NOT NULL DEFAULT '[]',
                    turn_history TEXT NOT NULL DEFAULT '[]',
                    state TEXT NOT NULL DEFAULT '{}',
                    config TEXT NOT NULL DEFAULT '{}',
                    metadata TEXT NOT NULL DEFAULT '{}'
                )
            """)
            
            # 创建索引
            await db.execute("""
                CREATE INDEX IF NOT EXISTS idx_sessions_status 
                ON sessions(status)
            """)
            await db.execute("""
                CREATE INDEX IF NOT EXISTS idx_sessions_updated 
                ON sessions(updated_at)
            """)
            
            await db.commit()
    
    async def create(self, session_id: str = None, config: dict = None,
                     metadata: dict = None) -> Session:
        await self._init_db()
        
        session = Session(
            session_id=session_id or str(uuid4()),
            config=config or {},
            metadata=metadata or {},
        )
        
        async with aiosqlite.connect(self.db_path) as db:
            await db.execute(
                """INSERT INTO sessions 
                   (session_id, status, created_at, updated_at, 
                    messages, turn_history, state, config, metadata)
                   VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)""",
                (
                    session.session_id,
                    session.status.value,
                    session.created_at.isoformat(),
                    session.updated_at.isoformat(),
                    json.dumps(session.messages),
                    json.dumps(session.turn_history),
                    json.dumps(session.state),
                    json.dumps(session.config),
                    json.dumps(session.metadata),
                )
            )
            await db.commit()
        
        return session
    
    async def get(self, session_id: str) -> Optional[Session]:
        await self._init_db()
        
        async with aiosqlite.connect(self.db_path) as db:
            db.row_factory = aiosqlite.Row
            cursor = await db.execute(
                "SELECT * FROM sessions WHERE session_id = ?",
                (session_id,)
            )
            row = await cursor.fetchone()
        
        if row is None:
            return None
        
        return Session(
            session_id=row["session_id"],
            status=SessionStatus(row["status"]),
            created_at=datetime.fromisoformat(row["created_at"]),
            updated_at=datetime.fromisoformat(row["updated_at"]),
            messages=json.loads(row["messages"]),
            turn_history=json.loads(row["turn_history"]),
            state=json.loads(row["state"]),
            config=json.loads(row["config"]),
            metadata=json.loads(row["metadata"]),
        )
    
    async def save(self, session: Session) -> None:
        await self._init_db()
        
        session.updated_at = datetime.now()
        
        async with aiosqlite.connect(self.db_path) as db:
            await db.execute(
                """UPDATE sessions SET
                   status = ?, updated_at = ?, messages = ?,
                   turn_history = ?, state = ?, config = ?, metadata = ?
                   WHERE session_id = ?""",
                (
                    session.status.value,
                    session.updated_at.isoformat(),
                    json.dumps(session.messages),
                    json.dumps(session.turn_history),
                    json.dumps(session.state),
                    json.dumps(session.config),
                    json.dumps(session.metadata),
                    session.session_id,
                )
            )
            await db.commit()
    
    async def append_message(self, session_id: str, message: dict) -> None:
        await self._init_db()
        
        async with aiosqlite.connect(self.db_path) as db:
            # 读取当前 messages
            cursor = await db.execute(
                "SELECT messages FROM sessions WHERE session_id = ?",
                (session_id,)
            )
            row = await cursor.fetchone()
            if row is None:
                raise ValueError(f"Session {session_id} not found")
            
            messages = json.loads(row[0])
            messages.append(message)
            
            await db.execute(
                "UPDATE sessions SET messages = ?, updated_at = ? WHERE session_id = ?",
                (json.dumps(messages), datetime.now().isoformat(), session_id)
            )
            await db.commit()
    
    async def update_status(self, session_id: str, status: SessionStatus) -> None:
        await self._init_db()
        
        async with aiosqlite.connect(self.db_path) as db:
            await db.execute(
                "UPDATE sessions SET status = ?, updated_at = ? WHERE session_id = ?",
                (status.value, datetime.now().isoformat(), session_id)
            )
            await db.commit()
    
    async def delete(self, session_id: str) -> None:
        await self._init_db()
        
        async with aiosqlite.connect(self.db_path) as db:
            await db.execute(
                "DELETE FROM sessions WHERE session_id = ?",
                (session_id,)
            )
            await db.commit()
    
    async def list_sessions(self, limit: int = 100, offset: int = 0) -> list[Session]:
        await self._init_db()
        
        async with aiosqlite.connect(self.db_path) as db:
            db.row_factory = aiosqlite.Row
            cursor = await db.execute(
                "SELECT * FROM sessions ORDER BY updated_at DESC LIMIT ? OFFSET ?",
                (limit, offset)
            )
            rows = await cursor.fetchall()
        
        return [
            Session(
                session_id=row["session_id"],
                status=SessionStatus(row["status"]),
                created_at=datetime.fromisoformat(row["created_at"]),
                updated_at=datetime.fromisoformat(row["updated_at"]),
                messages=json.loads(row["messages"]),
                turn_history=json.loads(row["turn_history"]),
                state=json.loads(row["state"]),
                config=json.loads(row["config"]),
                metadata=json.loads(row["metadata"]),
            )
            for row in rows
        ]
```

### 2.3 Redis 实现 —— 高速缓存层

Redis 适合以下场景：
- 需要高速读写（微秒级延迟）
- 会话数据不需要长期持久化（内存数据库，宕机会丢数据）
- 需要 TTL 自动过期（不需要手动清理过期会话）
- 轻量级数据存储（单个 session 数据不宜过大，建议 < 10MB）

```python
# framework/session/redis_store.py
"""Redis 会话存储引擎"""

from __future__ import annotations
import json
from datetime import datetime
from typing import Any, Optional
from uuid import uuid4

from redis.asyncio import Redis

from .base import Session, SessionStatus, SessionStore


class RedisSessionStore(SessionStore):
    """Redis 实现的会话存储
    
    优势：高速读写（微秒级）、TTL 自动过期、
          适合高并发 session 读写场景
    局限：内存有限、持久化需额外配置（RDB/AOF）、
          存储大数据时成本高
    """
    
    def __init__(
        self,
        redis: Redis,
        ttl_seconds: int = 3600,       # 1小时无访问自动过期
        session_prefix: str = "agent:session:",
    ):
        self.redis = redis
        self.ttl = ttl_seconds
        self.prefix = session_prefix
    
    def _key(self, session_id: str) -> str:
        return f"{self.prefix}{session_id}"
    
    def _index_key(self) -> str:
        return f"{self.prefix}index"
    
    async def create(self, session_id: str = None, config: dict = None,
                     metadata: dict = None) -> Session:
        session = Session(
            session_id=session_id or str(uuid4()),
            config=config or {},
            metadata=metadata or {},
        )
        
        # 存储为 Hash
        await self.redis.hset(self._key(session.session_id), mapping={
            "session_id": session.session_id,
            "status": session.status.value,
            "created_at": session.created_at.isoformat(),
            "updated_at": session.updated_at.isoformat(),
            "messages": json.dumps(session.messages),
            "turn_history": json.dumps(session.turn_history),
            "state": json.dumps(session.state),
            "config": json.dumps(session.config),
            "metadata": json.dumps(session.metadata),
        })
        await self.redis.expire(self._key(session.session_id), self.ttl)
        
        # 维护索引
        await self.redis.zadd(self._index_key(), {
            session.session_id: session.created_at.timestamp()
        })
        
        return session
    
    async def get(self, session_id: str) -> Optional[Session]:
        data = await self.redis.hgetall(self._key(session_id))
        if not data:
            return None
        
        # Redis 返回 bytes，需要解码
        decoded = {k.decode(): v.decode() if isinstance(v, bytes) else v 
                   for k, v in data.items()}
        
        session = Session(
            session_id=decoded["session_id"],
            status=SessionStatus(decoded["status"]),
            created_at=datetime.fromisoformat(decoded["created_at"]),
            updated_at=datetime.fromisoformat(decoded["updated_at"]),
            messages=json.loads(decoded["messages"]),
            turn_history=json.loads(decoded["turn_history"]),
            state=json.loads(decoded["state"]),
            config=json.loads(decoded["config"]),
            metadata=json.loads(decoded["metadata"]),
        )
        
        # 刷新 TTL
        await self.redis.expire(self._key(session_id), self.ttl)
        
        return session
    
    async def save(self, session: Session) -> None:
        session.updated_at = datetime.now()
        
        await self.redis.hset(self._key(session.session_id), mapping={
            "session_id": session.session_id,
            "status": session.status.value,
            "updated_at": session.updated_at.isoformat(),
            "messages": json.dumps(session.messages),
            "turn_history": json.dumps(session.turn_history),
            "state": json.dumps(session.state),
            "config": json.dumps(session.config),
            "metadata": json.dumps(session.metadata),
        })
        await self.redis.expire(self._key(session.session_id), self.ttl)
    
    async def append_message(self, session_id: str, message: dict) -> None:
        key = self._key(session_id)
        
        # 使用 Redis Lua 脚本原子操作
        lua_script = """
            local data = redis.call('HGET', KEYS[1], 'messages')
            if not data then
                return nil
            end
            local messages = cjson.decode(data)
            table.insert(messages, cjson.decode(ARGV[1]))
            redis.call('HSET', KEYS[1], 'messages', cjson.encode(messages))
            redis.call('HSET', KEYS[1], 'updated_at', ARGV[2])
            redis.call('EXPIRE', KEYS[1], ARGV[3])
            return 1
        """
        
        result = await self.redis.eval(
            lua_script,
            keys=[key],
            args=[
                json.dumps(message),
                datetime.now().isoformat(),
                str(self.ttl),
            ]
        )
        
        if result is None:
            raise ValueError(f"Session {session_id} not found")
    
    async def update_status(self, session_id: str, status: SessionStatus) -> None:
        key = self._key(session_id)
        exists = await self.redis.exists(key)
        if not exists:
            raise ValueError(f"Session {session_id} not found")
        
        await self.redis.hset(key, "status", status.value)
        await self.redis.hset(key, "updated_at", datetime.now().isoformat())
        await self.redis.expire(key, self.ttl)
    
    async def delete(self, session_id: str) -> None:
        await self.redis.delete(self._key(session_id))
        await self.redis.zrem(self._index_key(), session_id)
    
    async def list_sessions(self, limit: int = 100, offset: int = 0) -> list[Session]:
        # 从有序集合中获取最近的 session ID
        session_ids = await self.redis.zrevrange(
            self._index_key(), offset, offset + limit - 1
        )
        
        sessions = []
        for sid in session_ids:
            session = await self.get(sid.decode())
            if session:
                sessions.append(session)
        
        return sessions
```

### 2.4 PostgreSQL 实现 —— 生产级持久化

PostgreSQL 适合以下场景：
- 生产环境，需要 ACID 保证
- 数据量大（GB 到 TB 级别）
- 需要复杂查询（按时间、状态、用户过滤）
- 多个服务共享会话数据
- 需要备份和恢复能力

```python
# framework/session/postgres_store.py
"""PostgreSQL 会话存储引擎"""

from __future__ import annotations
import json
from datetime import datetime
from typing import Any, Optional
from uuid import uuid4

from asyncpg import Pool, create_pool

from .base import Session, SessionStatus, SessionStore


class PostgresSessionStore(SessionStore):
    """PostgreSQL 实现的会话存储
    
    优势：ACID 保证、复杂查询能力、高可用配置、
          大容量存储、适合生产环境
    局限：需要独立数据库服务、延迟比 Redis 高（1-5ms）
    """
    
    def __init__(
        self,
        pool: Pool,
        table_name: str = "agent_sessions",
    ):
        self.pool = pool
        self.table = table_name
    
    async def _init_db(self):
        """初始化表结构"""
        async with self.pool.acquire() as conn:
            await conn.execute(f"""
                CREATE TABLE IF NOT EXISTS {self.table} (
                    session_id UUID PRIMARY KEY,
                    status VARCHAR(20) NOT NULL DEFAULT 'active',
                    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
                    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
                    messages JSONB NOT NULL DEFAULT '[]',
                    turn_history JSONB NOT NULL DEFAULT '[]',
                    state JSONB NOT NULL DEFAULT '{{}}',
                    config JSONB NOT NULL DEFAULT '{{}}',
                    metadata JSONB NOT NULL DEFAULT '{{}}'
                )
            """)
            
            # 创建索引
            await conn.execute(f"""
                CREATE INDEX IF NOT EXISTS idx_{self.table}_status
                ON {self.table}(status)
            """)
            await conn.execute(f"""
                CREATE INDEX IF NOT EXISTS idx_{self.table}_updated
                ON {self.table}(updated_at DESC)
            """)
            await conn.execute(f"""
                CREATE INDEX IF NOT EXISTS idx_{self.table}_metadata
                ON {self.table} USING GIN(metadata)
            """)
    
    async def create(self, session_id: str = None, config: dict = None,
                     metadata: dict = None) -> Session:
        await self._init_db()
        
        session = Session(
            session_id=session_id or str(uuid4()),
            config=config or {},
            metadata=metadata or {},
        )
        
        async with self.pool.acquire() as conn:
            await conn.execute(
                f"""INSERT INTO {self.table} 
                   (session_id, status, created_at, updated_at,
                    messages, turn_history, state, config, metadata)
                   VALUES ($1, $2, $3, $4, $5::jsonb, $6::jsonb,
                           $7::jsonb, $8::jsonb, $9::jsonb)""",
                session.session_id,
                session.status.value,
                session.created_at,
                session.updated_at,
                json.dumps(session.messages),
                json.dumps(session.turn_history),
                json.dumps(session.state),
                json.dumps(session.config),
                json.dumps(session.metadata),
            )
        
        return session
    
    async def get(self, session_id: str) -> Optional[Session]:
        await self._init_db()
        
        async with self.pool.acquire() as conn:
            row = await conn.fetchrow(
                f"SELECT * FROM {self.table} WHERE session_id = $1",
                session_id
            )
        
        if row is None:
            return None
        
        return Session(
            session_id=str(row["session_id"]),
            status=SessionStatus(row["status"]),
            created_at=row["created_at"],
            updated_at=row["updated_at"],
            messages=row["messages"],
            turn_history=row["turn_history"],
            state=row["state"],
            config=row["config"],
            metadata=row["metadata"],
        )
    
    async def save(self, session: Session) -> None:
        session.updated_at = datetime.now()
        
        async with self.pool.acquire() as conn:
            await conn.execute(
                f"""UPDATE {self.table} SET
                   status = $1, updated_at = $2,
                   messages = $3::jsonb, turn_history = $4::jsonb,
                   state = $5::jsonb, config = $6::jsonb,
                   metadata = $7::jsonb
                   WHERE session_id = $8""",
                session.status.value,
                session.updated_at,
                json.dumps(session.messages),
                json.dumps(session.turn_history),
                json.dumps(session.state),
                json.dumps(session.config),
                json.dumps(session.metadata),
                session.session_id,
            )
    
    async def append_message(self, session_id: str, message: dict) -> None:
        await self._init_db()
        
        async with self.pool.acquire() as conn:
            # PostgreSQL JSONB 支持直接追加
            result = await conn.execute(
                f"""UPDATE {self.table} SET
                   messages = messages || $1::jsonb,
                   updated_at = NOW()
                   WHERE session_id = $2""",
                json.dumps([message]),
                session_id,
            )
            
            if result == "UPDATE 0":
                raise ValueError(f"Session {session_id} not found")
    
    async def update_status(self, session_id: str, status: SessionStatus) -> None:
        await self._init_db()
        
        async with self.pool.acquire() as conn:
            await conn.execute(
                f"UPDATE {self.table} SET status = $1, updated_at = NOW() WHERE session_id = $2",
                status.value, session_id
            )
    
    async def delete(self, session_id: str) -> None:
        await self._init_db()
        
        async with self.pool.acquire() as conn:
            await conn.execute(
                f"DELETE FROM {self.table} WHERE session_id = $1",
                session_id
            )
    
    async def list_sessions(self, limit: int = 100, offset: int = 0) -> list[Session]:
        await self._init_db()
        
        async with self.pool.acquire() as conn:
            rows = await conn.fetch(
                f"SELECT * FROM {self.table} ORDER BY updated_at DESC LIMIT $1 OFFSET $2",
                limit, offset
            )
        
        return [
            Session(
                session_id=str(row["session_id"]),
                status=SessionStatus(row["status"]),
                created_at=row["created_at"],
                updated_at=row["updated_at"],
                messages=row["messages"],
                turn_history=row["turn_history"],
                state=row["state"],
                config=row["config"],
                metadata=row["metadata"],
            )
            for row in rows
        ]
```

## 四、Resume（会话恢复）机制

### 4.1 恢复流程

Resume 机制允许 Agent 在崩溃、重启或用户离开后，无缝恢复之前的会话状态。这是生产级 Agent 系统的关键能力。

```
Resume 流程：
1. 客户端提供 session_id
2. 会话管理器从存储中加载 Session
3. 检查会话状态（ACTIVE → 继续；ENDED → 报错）
4. 重建对话上下文（messages[]）
5. 重建 Agent 状态（当前步骤、中间变量）
6. 返回重建后的 Context，Agent 继续执行
```

### 4.2 Resume 实现

```python
# framework/session/resume.py
"""会话恢复机制"""

from __future__ import annotations
import logging
from typing import Optional

from .base import Session, SessionStatus, SessionStore
from ..memory import Memory

logger = logging.getLogger(__name__)


class SessionManager:
    """会话管理器 —— 管理和恢复 Agent 会话"""
    
    def __init__(
        self,
        store: SessionStore,
        memory: Memory,
    ):
        self.store = store
        self.memory = memory
    
    async def create_session(
        self,
        session_id: str = None,
        system_prompt: str = None,
        model: str = None,
        temperature: float = 0.7,
        metadata: dict = None,
    ) -> Session:
        """创建一个新会话"""
        config = {}
        if system_prompt:
            config["system_prompt"] = system_prompt
        if model:
            config["model"] = model
        if temperature is not None:
            config["temperature"] = temperature
        
        session = await self.store.create(
            session_id=session_id,
            config=config,
            metadata=metadata or {},
        )
        
        logger.info(f"Created session: {session.session_id}")
        return session
    
    async def resume_session(
        self,
        session_id: str,
    ) -> Session:
        """恢复一个已存在的会话
        
        恢复流程：
        1. 从存储中加载会话
        2. 检查会话状态
        3. 重建记忆上下文
        4. 返回恢复后的会话
        """
        session = await self.store.get(session_id)
        
        if session is None:
            raise SessionNotFoundError(f"Session {session_id} not found")
        
        if session.status == SessionStatus.ENDED:
            raise SessionEndedError(f"Session {session_id} has already ended")
        
        # 重建记忆
        await self.memory.clear()
        for msg in session.messages:
            await self.memory.add(
                role=msg.get("role", "user"),
                content=msg.get("content", ""),
                metadata=msg.get("metadata", {}),
            )
        
        # 如果会话是 PAUSED，恢复为 ACTIVE
        if session.status == SessionStatus.PAUSED:
            session.status = SessionStatus.ACTIVE
            await self.store.save(session)
        
        logger.info(
            f"Resumed session: {session_id} "
            f"(messages={len(session.messages)}, "
            f"turns={len(session.turn_history)})"
        )
        
        return session
    
    async def pause_session(self, session_id: str) -> None:
        """暂停会话（保存状态但不结束）"""
        session = await self.store.get(session_id)
        if session is None:
            raise SessionNotFoundError(f"Session {session_id} not found")
        
        # 保存当前记忆到会话状态
        context = await self.memory.get_context()
        session.state["memory_snapshot"] = [
            {"role": m.role, "content": m.content}
            for m in context
        ]
        session.state["memory_tokens"] = sum(
            len(m.content) // 4 for m in context
        )
        
        session.status = SessionStatus.PAUSED
        await self.store.save(session)
        logger.info(f"Paused session: {session_id}")
    
    async def end_session(self, session_id: str) -> None:
        """结束会话"""
        await self.store.update_status(session_id, SessionStatus.ENDED)
        logger.info(f"Ended session: {session_id}")
    
    async def record_turn(self, session_id: str, turn_data: dict) -> None:
        """记录一个 Turn 到会话历史"""
        session = await self.store.get(session_id)
        if session is None:
            return
        
        session.turn_history.append(turn_data)
        session.updated_at = turn_data.get("timestamp", session.updated_at)
        
        await self.store.save(session)


class SessionNotFoundError(Exception):
    """会话不存在"""
    pass

class SessionEndedError(Exception):
    """会话已结束"""
    pass
```

## 五、存储引擎的迁移策略

### 5.1 从 SQLite 迁移到 PostgreSQL

项目从开发到上线，通常经历 SQLite → PostgreSQL 的迁移过程：

```python
# framework/session/migration.py
"""存储引擎迁移工具"""

import asyncio
import json
from framework.session.sqlite_store import SQLiteSessionStore
from framework.session.postgres_store import PostgresSessionStore


async def migrate_sqlite_to_postgres(
    sqlite_path: str,
    pg_pool,
    batch_size: int = 100,
):
    """将 SQLite 数据迁移到 PostgreSQL"""
    sqlite_store = SQLiteSessionStore(sqlite_path)
    pg_store = PostgresSessionStore(pg_pool)
    
    offset = 0
    total = 0
    
    while True:
        sessions = await sqlite_store.list_sessions(
            limit=batch_size, offset=offset
        )
        if not sessions:
            break
        
        for session in sessions:
            try:
                # 在 PostgreSQL 中创建相同数据
                await pg_store.create(
                    session_id=session.session_id,
                    config=session.config,
                    metadata=session.metadata,
                )
                
                # 恢复消息
                for msg in session.messages:
                    await pg_store.append_message(
                        session.session_id, msg
                    )
                
                # 更新状态
                await pg_store.update_status(
                    session.session_id, session.status
                )
                
                total += 1
                print(f"Migrated session: {session.session_id}")
                
            except Exception as e:
                print(f"Failed to migrate {session.session_id}: {e}")
        
        offset += batch_size
    
    print(f"Migration complete: {total} sessions migrated")
```

### 5.2 多级存储策略

生产环境中，可以使用 Redis + PostgreSQL 组合：

- **Redis**：存储"热"会话（最近 30 分钟内活跃的）
- **PostgreSQL**：存储所有会话（全量数据）
- **LRU 淘汰**：Redis 写满时，最不活跃的会话自动淘汰到 PostgreSQL

```python
class TieredSessionStore(SessionStore):
    """分层会话存储 —— Redis(热) + PostgreSQL(冷)"""
    
    async def get(self, session_id: str) -> Optional[Session]:
        # 先查 Redis（热数据）
        session = await self.hot_store.get(session_id)
        if session:
            return session
        
        # 未命中，查 PostgreSQL（冷数据）
        session = await self.cold_store.get(session_id)
        if session:
            # 预热到 Redis
            await self.hot_store.create(
                session_id=session.session_id,
                config=session.config,
                metadata=session.metadata,
            )
        
        return session
```
